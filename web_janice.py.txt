# 2. The function that runs when you click "Start Debate"
def ask_janice(question):
    try:
        env = os.environ.copy()
        result = subprocess.run(
            ["parliament", "ask", question],
            capture_output=True,
            text=True,
            env=env,
            timeout=300
        )
        if result.returncode == 0:
            return result.stdout
        else:
            return f"Parliament Error:\n{result.stderr}\n\nOutput:\n{result.stdout}"
    except Exception as e:
        return f"Unexpected Error: {str(e)}"

# 3. Build the Web Interface
with gr.Blocks(title="Janice AI") as demo:
    gr.Markdown("# 🧠 Janice: Multi-Agent Debate Panel")
    gr.Markdown("Ask a question. Janice will have her models debate it and synthesize a final answer.")
    
    with gr.Row():
        inp = gr.Textbox(label="Your Question", placeholder="Should I expand my business?")
    with gr.Row():
        btn = gr.Button("Start Debate", variant="primary")
    with gr.Row():
        out = gr.Textbox(label="Janice's Verdict", lines=30)
        
    btn.click(fn=ask_janice, inputs=inp, outputs=out)

# 4. Launch the app on Render's required port
if __name__ == "__main__":
    port = int(os.environ.get("PORT", 7860))
    demo.launch(server_name="0.0.0.0", server_port=port)
